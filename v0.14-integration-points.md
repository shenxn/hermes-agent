### _resolve_draft_streaming (lines 834–876)

```python
def _resolve_draft_streaming(self) -> bool:
    """Decide whether this run should use native draft streaming."""
    transport = (self.cfg.transport or "auto").lower()
    if transport == "edit":
        return False
    if transport == "off":
        return False
    if not isinstance(self.adapter, _BasePlatformAdapter):
        return False
    try:
        supported = self.adapter.supports_draft_streaming(
            chat_type=self.cfg.chat_type or None,
            metadata=self.metadata,
        )
    except Exception:
        logger.debug("supports_draft_streaming probe raised", exc_info=True)
        supported = False
    if not supported:
        if transport == "draft":
            logger.debug(
                "Draft streaming requested but unsupported (chat=%s, type=%r) — "
                "falling back to edit",
                self.chat_id, self.cfg.chat_type,
            )
        return False
    return True
```

```python
async def _send_or_edit(self, text: str, *, finalize: bool = False) -> bool:
    """Send or edit the streaming message."""
    text = self._clean_for_display(text)
    visible_without_cursor = text
    if self.cfg.cursor:
        visible_without_cursor = visible_without_cursor.replace(self.cfg.cursor, "")
    _visible_stripped = visible_without_cursor.strip()
    if not _visible_stripped:
        return True
    if not text.strip():
        return True

    _MIN_NEW_MSG_CHARS = 4
    if (self._message_id is None
            and self.cfg.cursor
            and self.cfg.cursor in text
            and len(_visible_stripped) < _MIN_NEW_MSG_CHARS):
        return True

    # Native draft streaming: route mid-stream frames through send_draft.
    if (
        self._use_draft_streaming
        and not finalize
        and self._message_id is None
    ):
        if text == self._last_sent_text:
            return True
        ok = await self._send_draft_frame(text)
        if ok:
            return True
        # Failure already disabled drafts; fall through to edit path.

    try:
        if self._message_id is not None:
            if self._edit_supported:
                if text == self._last_sent_text and not (
                    finalize and self._adapter_requires_finalize
                ):
                    return True
                if (
                    finalize
                    and self._should_send_fresh_final()
                    and await self._try_fresh_final(text)
                ):
                    return True
                # Edit existing message
                result = await self.adapter.edit_message(
                    chat_id=self.chat_id,
                    message_id=self._message_id,
                    content=text,
                    finalize=finalize,
                )
                if result.success:
                    self._already_sent = True
                    _continuation_ids = getattr(result, "continuation_message_ids", ()) or ()
                    if (
                        _continuation_ids
                        and result.message_id
                        and result.message_id != self._message_id
                    ):
                        self._message_id = str(result.message_id)
                        self._message_created_ts = time.monotonic()
                        self._last_sent_text = ""
                        self._notify_new_message()
                    else:
                        self._last_sent_text = text
                    # ... flood control handling follows
```

### edit_message() (lines 1814–1863)

```python
async def edit_message(
    self,
    chat_id: str,
    message_id: str,
    content: str,
    *,
    finalize: bool = False,
) -> SendResult:
    """Edit a previously sent Feishu text/post message."""
    if not self._client:
        return SendResult(success=False, error="Not connected")

    content = self.format_message(content)
    try:
        msg_type, payload = self._build_outbound_payload(content)
        body = self._build_update_message_body(msg_type=msg_type, content=payload)
        request = self._build_update_message_request(message_id=message_id, request_body=body)
        response = await asyncio.to_thread(self._client.im.v1.message.update, request)
        result = self._finalize_send_result(response, "update failed")
        if not result.success and msg_type == "post" and _POST_CONTENT_INVALID_RE.search(result.error or ""):
            logger.warning("[Feishu] Invalid post update payload rejected by API; falling back to plain text")
            fallback_body = self._build_update_message_body(
                msg_type="text",
                content=json.dumps({"text": _strip_markdown_to_plain_text(content)}, ensure_ascii=False),
            )
            fallback_request = self._build_update_message_request(message_id=message_id, request_body=fallback_body)
            fallback_response = await asyncio.to_thread(self._client.im.v1.message.update, fallback_request)
            result = self._finalize_send_result(fallback_response, "update failed")
        if result.success:
            result.message_id = message_id
        return result
    except Exception as exc:
        logger.error("[Feishu] Failed to edit message %s: %s", message_id, exc, exc_info=True)
        return SendResult(success=False, error=str(exc))
```

### send_progress_messages — async consumer of progress_queue (lines 14797–14920+)

This is a long async function that:
1. Checks adapter supports `edit_message`
2. Loops on `progress_queue.get_nowait()`
3. First tool: `adapter.send()` to create the progress message
4. Subsequent tools: `adapter.edit_message()` to update it
5. Handles `__reset__` markers (from stream consumer `on_new_message` callback)
6. Falls back to new sends on edit failure (flood control)

---

## Key Architectural Notes for Patching

1. **Streaming pipeline**: `StreamConsumerConfig` → `GatewayStreamConsumer` → `adapter.edit_message()`.
   Feishu goes through the generic path — no Feishu-specific streaming state.

2. **Draft streaming** is resolved per-run via `_resolve_draft_streaming()`. Feishu
   would need `supports_draft_streaming()` on its adapter to participate.

3. **Tool progress** is a separate pipeline: `progress_callback` → `progress_queue`
   → `send_progress_messages()` async task → `adapter.edit_message()`.

4. **Status callback** (`_status_callback_sync`) bridges sync agent thread to async
   adapter.send(). Assigned as `agent.status_callback`.

5. **`stream_consumer_holder[0]`** is the mutable container pattern used to share
   the consumer between the sync `run_sync()` thread and the async `_start_stream_consumer()` task.

6. **`on_new_message` callback** on the stream consumer is used to linearize
   tool-progress bubbles with content bubbles. Set to `lambda: progress_queue.put(("__reset__",))`.
